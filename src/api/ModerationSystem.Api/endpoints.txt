UserController ('/users')
- GET:/ getAll() -> UserResponse[]
- GET:/${userId} getUser() -> UserResponse
- GET:/search?q=... searchUser() -> UserResponse[]
- GET:/${userId}/followers getFollowers() -> UserResponse[]
- GET:/${userId}/following getFollowing() -> UserResponse[]
- PATCH:/${id} updateUser(body: {displayName?: string, userName?: string, bio?: string}) -> UserResponse
- POST:/${userId}/follow followUser() -> void // follow/unfollow
- DELETE:/${userId} deleteUser() -> void

UserResponse = {
  id: string;
  userName: string;
  userId: string;
  createdAt: string;
  content: string;
  userAvatarUrl: string;
  isLiked: boolean;
  following: number;
  followers: number;
  youFollow: boolean;
}

PostsController ('/posts')
- GET:/?page=1&pageSize=10 getFeed() -> PostResponse[]
- GET:/${postId} getById() -> PostResponse
- GET:/api/users/${userId}/posts getUsersPosts() -> PostResponse[]
- POST:/ createPost(body: {content: string, parentPostId?: string}) -> PostResponse
- GET:/${id}/logs getLogs() -> PostLogResponse[]
- POST:/${id}/likes likePost() -> void
- DELETE:/${id}/likes likePost() -> void
- POST:/${id}/reviews postReview(body: {eviewType: ReviewType, description?: string}) -> ReviewResponse
- GET:/${id}/reviews getReviews() -> ReviewResponse[]

PostResponse = {
  id: string;
  userName: string;
  userId: string;
  createdAt: string;
  content: string;
  userAvatarUrl: string;
  isLiked: boolean;
  status: PostStatus;
  likeCount: number;
  commentCount: number;
}

PostLogResponse = {
  id: string;
  oldStatus: PostStatus;
  newStatus: PostStatus;
  reason: string;
  triggeredBy: "llm" | "moderator" | "community" | "system";
  createdAt: string;
}

ReviewType =
  | "InappropriateContent"
  | "Spam"
  | "Harassment"
  | "HateSpeech"
  | "Misinformation"
  | "Other";

AuthController ('/auth')
- POST:/sign-up signUp(body: {username: string, email: string, password: string, displayName?: string, bio?: string}) -> string
- POST:/confirm-email confirmEmail(body: {username: string, confirmationCode: string}) -> string
- POST:/log-in logIn(body: {username: string, password: string}) -> TokenResponse
- POST:/log-out logOut() -> void
- POST:/refresh-token?refreshToken=... refreshToken() -> TokenResponse

TokenResponse = {
  accessToken: string
  idToken: string // idk maybe smth from cognito
  refreshToken: string
  expiresIn: number
}
